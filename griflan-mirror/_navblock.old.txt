<script>(function(){
  var tries=0;
  var t=setInterval(function(){
    var holder=document.querySelector(".js-i-logo-post-fade");
    if(holder && window.PlanCCat && !holder.dataset.plancDone){
      holder.dataset.plancDone="1"; clearInterval(t);
      var cv=holder.querySelector("canvas"); if(cv) cv.style.display="none";
      var box=document.createElement("div"); box.style.cssText="position:absolute;inset:0";
      holder.appendChild(box);
      var cat=window.PlanCCat.mount(box,{coat:"#f2efde",zzz:false,sleepAfter:1e9});
      var link=holder.closest("a")||holder.parentElement;
      if(link){ link.addEventListener("mouseenter",function(){cat.hover(true);});
        link.addEventListener("mouseleave",function(){cat.hover(false);});
        link.addEventListener("click",function(){cat.click();}); }
      window.__planCNavCat=cat;
    }
    if(++tries>100) clearInterval(t);
  },100);
})();</script>
